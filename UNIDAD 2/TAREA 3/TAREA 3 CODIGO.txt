import numpy as np
import random
import math

def generar_estado_inicial(N=8):

    return np.random.permutation(N)  # Devuelve una permutación aleatoria de [0, 1, ..., N-1]

def calcular_conflictos(tablero):

    conflictos = 0
    N = len(tablero)
    
    for i in range(N):
        for j in range(i + 1, N):  # Comparar solo con las siguientes filas para evitar duplicados
            if abs(tablero[i] - tablero[j]) == abs(i - j):
                conflictos += 1

    return conflictos 

def generar_vecino(tablero):
    vecino = tablero.copy()  # Hacemos una copia del tablero actual
    i, j = random.sample(range(len(tablero)), 2)  # Seleccionamos dos filas aleatorias
    vecino[i], vecino[j] = vecino[j], vecino[i]  # Intercambiamos las columnas de esas filas
    
    return vecino  # Retorna el nuevo tablero modificado

def recocido_simulado(N=8, temperatura_inicial=1000, tasa_enfriamiento=0.99, iteraciones=10000):

    
    tablero = generar_estado_inicial(N)
    
    conflictos_actual = calcular_conflictos(tablero)
    
    temperatura = temperatura_inicial  

    for _ in range(iteraciones):
        if conflictos_actual == 0:
            return tablero  
        
        nuevo_tablero = generar_vecino(tablero)
        
        conflictos_nuevo = calcular_conflictos(nuevo_tablero)
        
        delta = conflictos_nuevo - conflictos_actual
        
        if delta < 0 or random.uniform(0, 1) < math.exp(-delta / temperatura):
            tablero = nuevo_tablero  
            conflictos_actual = conflictos_nuevo  
        
        temperatura *= tasa_enfriamiento  

        # Si la temperatura es muy baja, terminamos la búsqueda
        if temperatura < 1e-3:
            break


    return tablero if conflictos_actual == 0 else None


N = 8
solucion = recocido_simulado(N)

# Mostrar la solución
if solucion is not None:
    print("Solución encontrada:", solucion)
    for i in range(N):
        fila = ["Q" if solucion[i] == j else "." for j in range(N)]
        print(" ".join(fila))
else:
    print("No se encontró solución")
